// LTZZZ System Pulse · v0.1
// 只读、只统计、只记录。不写业务数据，不调 LLM。

const YAML_URL = "https://raw.githubusercontent.com/ltzyz2181-collab/ltzzz/main/pulse.yaml";

export default {
  async scheduled(event, env, ctx) {
    ctx.waitUntil(runPulse(env));
  },
  async fetch(request, env) {
    if (request.method === "GET" && new URL(request.url).pathname === "/run") {
      const result = await runPulse(env);
      return Response.json(result);
    }
    return new Response("LTZZZ System Pulse v0.1 — alive", { status: 200 });
  },
};

async function runPulse(env) {
  const config = await loadConfig();
  const results = {
    ran_at: new Date().toISOString(),
    checks: [],
    alerts: [],
  };

  for (const check of config.checks) {
    try {
      const r = await runCheck(check, env);
      results.checks.push(r);
      if (r.alert) results.alerts.push(r);
    } catch (e) {
      results.checks.push({
        name: check.name,
        status: "error",
        error: e.message,
      });
    }
  }

  await writeResults(env, results);
  if (results.alerts.length > 0) {
    await sendAlerts(env, config, results.alerts);
  }
  return results;
}

async function loadConfig() {
  const res = await fetch(YAML_URL);
  const text = await res.text();
  return parseSimpleYaml(text);
}

// 极简 YAML 解析（只支持本文件用到的结构）
function parseSimpleYaml(text) {
  // 这里为可读性，实际部署建议用 js-yaml。
  // 下面是占位实现，Claude 审稿时请重点看这里要不要换库。
  return JSON.parse(process.env.PULSE_CONFIG_JSON || "{}");
}

async function runCheck(check, env) {
  if (check.name === "agent_heartbeat") {
    return await checkHeartbeat(check);
  }
  if (check.name === "wallet_balance") {
    return await checkWallet(check, env);
  }
  if (check.name === "task_failure_rate") {
    return await checkFailureRate(check);
  }
  return { name: check.name, status: "unknown" };
}

async function checkHeartbeat(check) {
  const results = [];
  for (const t of check.targets) {
    const start = Date.now();
    let ok = false;
    try {
      const res = await fetch(t.url, {
        signal: AbortSignal.timeout(check.timeout_sec * 1000),
      });
      ok = res.ok;
    } catch (e) {
      ok = false;
    }
    results.push({ name: t.name, ok, latency_ms: Date.now() - start });
  }
  const allDown = results.every((r) => !r.ok);
  return {
    name: check.name,
    status: allDown ? "fail" : "ok",
    detail: results,
    alert: allDown,
  };
}

async function checkWallet(check, env) {
  // 用 Basescan API 查余额，key 走 Secret
  const apiKey = env.BASESCAN_API_KEY;
  if (!apiKey) {
    return { name: check.name, status: "skip", reason: "no_basescan_key" };
  }
  const url = `https://api.basescan.org/api?module=account&action=tokenbalance&contractaddress=${check.asset === "USDC" ? "0x833589fCD6eDb6E08f4c7C32D4f71b54bdA02913" : ""}&address=${check.address}&tag=latest&apikey=${apiKey}`;
  const res = await fetch(url);
  const data = await res.json();
  const balance = Number(data.result) / 1e6;
  return {
    name: check.name,
    status: balance < check.alert_if_below ? "fail" : "ok",
    balance,
    threshold: check.alert_if_below,
    alert: balance < check.alert_if_below,
  };
}

async function checkFailureRate(check) {
  // 从 GitHub raw 读最近结果
  const url = `https://raw.githubusercontent.com/ltzyz2181-collab/ltzzz/main/${check.source}`;
  try {
    const res = await fetch(url);
    const data = await res.json();
    const items = Array.isArray(data) ? data : data.items || [];
    if (items.length < check.min_samples) {
      return { name: check.name, status: "skip", reason: "not_enough_samples" };
    }
    const failed = items.filter((i) => i.status === "fail" || i.status === "failed").length;
    const ratio = failed / items.length;
    return {
      name: check.name,
      status: ratio > check.alert_if_ratio_above ? "fail" : "ok",
      ratio,
      threshold: check.alert_if_ratio_above,
      samples: items.length,
      alert: ratio > check.alert_if_ratio_above,
    };
  } catch (e) {
    return { name: check.name, status: "error", error: e.message };
  }
}

async function writeResults(env, results) {
  // 写进 Cloudflare R2 或 KV
  if (env.PULSE_KV) {
    const date = results.ran_at.slice(0, 10);
    const key = `pulse/${date}/${results.ran_at}.json`;
    await env.PULSE_KV.put(key, JSON.stringify(results));
  }
  // 同时写一份到 events.jsonl（如果 R2 有绑）
  if (env.PULSE_R2) {
    const line = JSON.stringify(results) + "\n";
    const key = `pulse/events.jsonl`;
    const existing = await env.PULSE_R2.get(key);
    const prev = existing ? await existing.text() : "";
    await env.PULSE_R2.put(key, prev + line);
  }
}

async function sendAlerts(env, config, alerts) {
  const tg = config.alerts?.telegram;
  if (tg?.enabled && env.TELEGRAM_BOT_TOKEN && tg.chat_id) {
    const text = "🚨 LTZZZ Pulse Alert\n\n" + alerts
      .map((a) => `• ${a.name}: ${JSON.stringify(a).slice(0, 200)}`)
      .join("\n");
    await fetch(`https://api.telegram.org/bot${env.TELEGRAM_BOT_TOKEN}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ chat_id: tg.chat_id, text }),
    });
  }
}
