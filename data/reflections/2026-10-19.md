async function claimDailyTask(taskId, reservationApi) {
  const res = await fetch(`${reservationApi}/claim`, {

    headers: { Authorization: `Bearer ${process.env.AGENT_RUN_TOKEN}` },
    body: JSON.stringify({ taskId, runId: process.env.AGENT_RUN_ID })
  });
  if (!res.ok) return { ok: false, reason: await res.text() };
  return { ok: true, claim: await res.json() };
}

async function runDailyTasks(tasks) {
  for (const task of tasks) {
    const claim = await claimDailyTask(task.id);
    if (!claim.ok) {
      await persistResult(task.id, { status: 'blocked', reason: claim.reason, ts: Date.now() });
      continue; // 断链修复：不可认领不执行，返回 blocked 并持久化
    }
    const result = await executeTask(task);
    await persistResult(task.id, result);
  }
}
